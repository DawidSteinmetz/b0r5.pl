-- ============================================================
-- 13. IMPORT — BEZPIECZNE ID I HIERARCHIA
-- ============================================================

DELIMITER $$

CREATE PROCEDURE hololink_import_mon_calamari()
BEGIN

    DECLARE v_planeta_id INT UNSIGNED DEFAULT NULL;
    DECLARE v_planeta_count INT DEFAULT 0;
    DECLARE v_next_id BIGINT UNSIGNED DEFAULT 0;
    DECLARE v_parent_id BIGINT UNSIGNED DEFAULT NULL;
    DECLARE v_nazwa VARCHAR(200);
    DECLARE v_typ VARCHAR(20);
    DECLARE v_rodzaj VARCHAR(20);
    DECLARE v_rodzic VARCHAR(200);
    DECLARE v_publiczna TINYINT;
    DECLARE v_status VARCHAR(20);
    DECLARE v_opis TEXT;
    DECLARE v_done TINYINT DEFAULT 0;
    DECLARE v_inserted INT DEFAULT 0;

    DECLARE cur CURSOR FOR
        SELECT nazwa,typ,rodzaj,rodzic,publiczna,status,opis
        FROM tmp_mon_calamari
        ORDER BY poziom,nazwa;

    DECLARE CONTINUE HANDLER FOR NOT FOUND
        SET v_done = 1;

    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        RESIGNAL;
    END;

    -- Weryfikacja planety.
    SELECT COUNT(*),MIN(id)
    INTO v_planeta_count,v_planeta_id
    FROM mapa_planety
    WHERE LOWER(TRIM(nazwa)) = LOWER('Mon Calamari');

    IF v_planeta_count <> 1 OR v_planeta_id <> 1459 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'Nie znaleziono dokładnie jednego rekordu Mon Calamari o ID 1459.';
    END IF;

    START TRANSACTION;

    -- ID nie ma AUTO_INCREMENT.
    SELECT COALESCE(MAX(id),0)+1
    INTO v_next_id
    FROM lokalizacje;

    OPEN cur;

    import_loop: LOOP

        FETCH cur INTO
            v_nazwa,
            v_typ,
            v_rodzaj,
            v_rodzic,
            v_publiczna,
            v_status,
            v_opis;

        IF v_done = 1 THEN
            LEAVE import_loop;
        END IF;

        -- Istniejący rekord pozostaje bez zmian.
        IF NOT EXISTS (
            SELECT 1
            FROM lokalizacje
            WHERE planeta_id = v_planeta_id
              AND nazwa = v_nazwa
        ) THEN

            SET v_parent_id = NULL;

            -- Wyszukiwanie rodzica po nazwie.
            IF v_rodzic IS NOT NULL THEN

                SET v_parent_id = (
                    SELECT id
                    FROM lokalizacje
                    WHERE planeta_id = v_planeta_id
                      AND nazwa = v_rodzic
                    LIMIT 1
                );

                IF v_parent_id IS NULL THEN
                    SIGNAL SQLSTATE '45000'
                    SET MESSAGE_TEXT =
                    'Brak rodzica dla lokalizacji — import przerwany.';
                END IF;

            END IF;

            -- Jawny przydział ID.
            INSERT INTO lokalizacje (
                id,
                typ,
                rodzaj,
                rodzic_id,
                uklad_id,
                planeta_id,
                nazwa,
                x_km,
                y_km,
                publiczna,
                status,
                opis
            )
            VALUES (
                v_next_id,
                v_typ,
                v_rodzaj,
                v_parent_id,
                NULL,
                v_planeta_id,
                v_nazwa,
                NULL,
                NULL,
                v_publiczna,
                v_status,
                v_opis
            );

            SET v_next_id = v_next_id + 1;
            SET v_inserted = v_inserted + 1;

        END IF;

    END LOOP;

    CLOSE cur;

    COMMIT;

    SELECT
        'IMPORT ZAKOŃCZONY' AS wynik,
        v_planeta_id AS planeta_id,
        v_inserted AS dodane_rekordy,
        (
            SELECT COUNT(*)
            FROM lokalizacje
            WHERE planeta_id = v_planeta_id
        ) AS wszystkie_lokalizacje_planety;

END$$

DELIMITER ;

CALL hololink_import_mon_calamari();

DROP PROCEDURE IF EXISTS hololink_import_mon_calamari;

-- ============================================================
-- 14. AUDYT PO IMPORCIE
-- ============================================================

SET @mon_calamari_id := 1459;

-- 14.1 Liczba wszystkich lokalizacji.
SELECT
    COUNT(*) AS wszystkie_lokalizacje
FROM lokalizacje
WHERE planeta_id = @mon_calamari_id;

-- 14.2 Liczba rekordów według typu.
SELECT
    typ,
    COUNT(*) AS liczba
FROM lokalizacje
WHERE planeta_id = @mon_calamari_id
GROUP BY typ
ORDER BY typ;

-- 14.3 Kontrola rodziców.
SELECT
    c.id,
    c.nazwa,
    c.rodzic_id,
    p.nazwa AS rodzic
FROM lokalizacje c
LEFT JOIN lokalizacje p
    ON p.id = c.rodzic_id
   AND p.planeta_id = c.planeta_id
WHERE c.planeta_id = @mon_calamari_id
  AND c.rodzic_id IS NOT NULL
  AND p.id IS NULL;

-- 14.4 Duplikaty nazw.
SELECT
    planeta_id,
    nazwa,
    COUNT(*) AS liczba
FROM lokalizacje
WHERE planeta_id = @mon_calamari_id
GROUP BY planeta_id,nazwa
HAVING COUNT(*) > 1;

-- 14.5 Korzenie geograficzne.
SELECT
    id,
    nazwa,
    typ,
    rodzic_id
FROM lokalizacje
WHERE planeta_id = @mon_calamari_id
  AND rodzic_id IS NULL
ORDER BY typ,nazwa;

-- 14.6 Kontrola Coral City.
SELECT
    id,
    nazwa,
    typ,
    status
FROM lokalizacje
WHERE planeta_id = @mon_calamari_id
  AND nazwa LIKE '%Coral City%';

-- 14.7 Kontrola miast Quarren.
SELECT
    id,
    nazwa,
    typ,
    rodzic_id,
    opis
FROM lokalizacje
WHERE planeta_id = @mon_calamari_id
  AND (
      nazwa LIKE '%Foamwander%'
      OR nazwa LIKE '%Quarren%'
      OR nazwa LIKE '%Morjanssik%'
  )
ORDER BY nazwa;

-- 14.8 Kontrola przemysłu stoczniowego.
SELECT
    id,
    nazwa,
    typ,
    rodzaj,
    rodzic_id,
    publiczna,
    status
FROM lokalizacje
WHERE planeta_id = @mon_calamari_id
  AND nazwa LIKE '%Shipyards%'
ORDER BY nazwa;

-- 14.9 Kontrola konkretnych biur stoczniowych.
SELECT
    id,
    nazwa,
    typ,
    rodzaj,
    publiczna,
    status
FROM lokalizacje
WHERE planeta_id = @mon_calamari_id
  AND nazwa LIKE '%Shipyards%'
  AND rodzaj = 'biuro'
ORDER BY nazwa;

-- 14.10 Kontrola oddzielenia orbity i powierzchni.
SELECT
    nazwa,
    typ,
    rodzic_id,
    opis
FROM lokalizacje
WHERE planeta_id = @mon_calamari_id
  AND nazwa IN (
      'Mon Calamari — orbita',
      'Mon Calamari — powierzchnia',
      'Mon Calamari Orbital Shipyards',
      'Mon Calamari Shipyards — kompleks planetarny'
  );

-- 14.11 Kontrola lokalizacji bez współrzędnych.
-- Oczekiwany wynik: większość nowych rekordów.
SELECT
    COUNT(*) AS rekordy_bez_wspolrzednych
FROM lokalizacje
WHERE planeta_id = @mon_calamari_id
  AND x_km IS NULL
  AND y_km IS NULL;

-- 14.12 Kontrola statusów.
SELECT
    status,
    COUNT(*) AS liczba
FROM lokalizacje
WHERE planeta_id = @mon_calamari_id
GROUP BY status;

-- 14.13 Pełne drzewo lokalizacji.
WITH RECURSIVE drzewo AS (
    SELECT
        id,
        nazwa,
        typ,
        rodzaj,
        rodzic_id,
        0 AS poziom,
        CAST(nazwa AS CHAR(4000)) AS sciezka
    FROM lokalizacje
    WHERE planeta_id = @mon_calamari_id
      AND rodzic_id IS NULL

    UNION ALL

    SELECT
        d.id,
        d.nazwa,
        d.typ,
        d.rodzaj,
        d.rodzic_id,
        t.poziom + 1,
        CONCAT(t.sciezka,' > ',d.nazwa)
    FROM lokalizacje d
    INNER JOIN drzewo t
        ON d.rodzic_id = t.id
    WHERE d.planeta_id = @mon_calamari_id
)
SELECT
    poziom,
    nazwa,
    typ,
    rodzaj,
    sciezka
FROM drzewo
ORDER BY sciezka;

-- 14.14 Kontrola obecności podstawowych elementów.
SELECT
    SUM(nazwa = 'Coral City') AS coral_city,
    SUM(nazwa = 'Foamwander City') AS foamwander_city,
    SUM(nazwa = 'Dac City') AS dac_city,
    SUM(nazwa = 'Mon Calamari Orbital Shipyards') AS orbital_shipyards,
    SUM(nazwa = 'Mon Calamari Shipyards — kompleks planetarny')
        AS planetary_shipyards,
    SUM(nazwa = 'Mon Calamari Shipyards — biura zarządu')
        AS management_offices,
    SUM(nazwa = 'Mon Calamari Shipyards — biuro projektowe')
        AS design_office,
    SUM(nazwa = 'Mon Calamari Shipyards — centrum inżynieryjne')
        AS engineering_center
FROM lokalizacje
WHERE planeta_id = @mon_calamari_id;

-- 14.15 Kontrola rekordów o nieprawidłowym typie.
SELECT
    id,
    nazwa,
    typ,
    rodzaj
FROM lokalizacje
WHERE planeta_id = @mon_calamari_id
  AND (
      typ NOT IN (
          'uklad','orbita','powierzchnia',
          'lokacja','budynek','ladowisko'
      )
      OR (
          rodzaj IS NOT NULL
          AND rodzaj NOT IN (
              'hangar','port','hotel','bankomat',
              'sklep','kantyna','biuro','garaz',
              'baza','magazyn','platforma','inne'
          )
      )
  );

-- 14.16 Kontrola osieroconych lokalizacji.
SELECT
    c.id,
    c.nazwa,
    c.rodzic_id
FROM lokalizacje c
LEFT JOIN lokalizacje p
    ON p.id = c.rodzic_id
WHERE c.planeta_id = @mon_calamari_id
  AND c.rodzic_id IS NOT NULL
  AND (
      p.id IS NULL
      OR p.planeta_id <> @mon_calamari_id
  );
