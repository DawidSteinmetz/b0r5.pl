    poziom TINYINT UNSIGNED NOT NULL,

DELIMITER $$

CREATE PROCEDURE hololink_import_kuat()
BEGIN
    DECLARE v_planeta_id INT UNSIGNED DEFAULT NULL;
    DECLARE v_planeta_count INT DEFAULT 0;

    DECLARE v_nazwa VARCHAR(200);
    DECLARE v_rodzic_nazwa VARCHAR(200);
    DECLARE v_typ VARCHAR(20);
    DECLARE v_rodzaj VARCHAR(20);
    DECLARE v_publiczna TINYINT;
    DECLARE v_status VARCHAR(20);
    DECLARE v_opis TEXT;

    DECLARE v_parent_id BIGINT UNSIGNED DEFAULT NULL;
    DECLARE v_id BIGINT UNSIGNED DEFAULT NULL;
    DECLARE v_existing_parent BIGINT UNSIGNED DEFAULT NULL;

    DECLARE v_done TINYINT DEFAULT 0;
    DECLARE v_before INT DEFAULT 0;
    DECLARE v_after INT DEFAULT 0;

    DECLARE c_lokacje CURSOR FOR
        SELECT
            nazwa,
            rodzic_nazwa,
            typ,
            rodzaj,
            publiczna,
            status,
            opis
        FROM tmp_kuat_lokacje
        ORDER BY poziom ASC, nazwa ASC;

    DECLARE CONTINUE HANDLER FOR NOT FOUND
        SET v_done = 1;

    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        RESIGNAL;
    END;

    SELECT COUNT(*), MIN(id)
      INTO v_planeta_count, v_planeta_id
      FROM mapa_planety
     WHERE LOWER(TRIM(nazwa)) = LOWER('Kuat');

    IF v_planeta_count <> 1 OR v_planeta_id <> 1262 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
        'BŁĄD: Kuat nie istnieje dokładnie raz pod oczekiwanym ID 1262.';
    END IF;

    SELECT COUNT(*)
      INTO v_before
      FROM lokalizacje
     WHERE planeta_id = v_planeta_id;

    START TRANSACTION;

    OPEN c_lokacje;

    import_loop: LOOP

        FETCH c_lokacje INTO
            v_nazwa,
            v_rodzic_nazwa,
            v_typ,
            v_rodzaj,
            v_publiczna,
            v_status,
            v_opis;

        IF v_done = 1 THEN
            LEAVE import_loop;
        END IF;

        SET v_parent_id = NULL;

        IF v_rodzic_nazwa IS NOT NULL THEN

            SET v_parent_id = (
                SELECT id
                  FROM lokalizacje
                 WHERE planeta_id = v_planeta_id
                   AND nazwa = v_rodzic_nazwa
                 LIMIT 1
            );

            IF v_parent_id IS NULL THEN
                SIGNAL SQLSTATE '45000'
                SET MESSAGE_TEXT =
                'BŁĄD: Nie znaleziono wymaganego rodzica lokalizacji.';
            END IF;

        END IF;

        INSERT INTO lokalizacje
        (
            typ,
            rodzaj,
            rodzic_id,
            uklad_id,
            planeta_id,
            nazwa,
            x_km,
            y_km,
            publiczna,
            status,
            opis
        )
        VALUES
        (
            v_typ,
            v_rodzaj,
            v_parent_id,
            NULL,
            v_planeta_id,
            v_nazwa,
            NULL,
            NULL,
            v_publiczna,
            v_status,
            v_opis
        )
        ON DUPLICATE KEY UPDATE
            id = LAST_INSERT_ID(id);

        SET v_id = LAST_INSERT_ID();

        SET v_existing_parent = (
            SELECT rodzic_id
              FROM lokalizacje
             WHERE id = v_id
             LIMIT 1
        );

        IF NOT (v_existing_parent <=> v_parent_id) THEN
            SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT =
            'BŁĄD: Istniejąca lokalizacja ma inną hierarchię rodzic-dziecko.';
        END IF;

    END LOOP;

    CLOSE c_lokacje;

    SELECT COUNT(*)
      INTO v_after
      FROM lokalizacje
     WHERE planeta_id = v_planeta_id;

    COMMIT;

    SELECT
        'IMPORT ZAKOŃCZONY POPRAWNIE' AS wynik,
        v_planeta_id AS planeta_id,
        v_before AS rekordy_przed_importem,
        v_after AS rekordy_po_imporcie,
        v_after - v_before AS liczba_dodanych_rekordow;

END$$

DELIMITER ;

CALL hololink_import_kuat();

DROP PROCEDURE IF EXISTS hololink_import_kuat;


-- ============================================================
-- AUDYT PO IMPORCIE
-- ============================================================

SET @kuat_planeta_id := 1262;


-- 1. ŁĄCZNA LICZBA LOKALIZACJI

SELECT
    COUNT(*) AS wszystkie_lokalizacje_kuat
FROM lokalizacje
WHERE planeta_id = @kuat_planeta_id;


-- 2. LICZBA LOKALIZACJI WEDŁUG TYPU

SELECT
    typ,
    COUNT(*) AS liczba
FROM lokalizacje
WHERE planeta_id = @kuat_planeta_id
GROUP BY typ
ORDER BY typ;


-- 3. LICZBA LOKALIZACJI WEDŁUG RODZAJU

SELECT
    rodzaj,
    COUNT(*) AS liczba
FROM lokalizacje
WHERE planeta_id = @kuat_planeta_id
GROUP BY rodzaj
ORDER BY rodzaj;


-- 4. KONTROLA OSIEROCONYCH REKORDÓW

SELECT
    dziecko.id,
    dziecko.nazwa,
    dziecko.rodzic_id
FROM lokalizacje dziecko
LEFT JOIN lokalizacje rodzic
    ON rodzic.id = dziecko.rodzic_id
WHERE dziecko.planeta_id = @kuat_planeta_id
  AND dziecko.rodzic_id IS NOT NULL
  AND rodzic.id IS NULL;


-- 5. KONTROLA DUPLIKATÓW NAZW

SELECT
    nazwa,
    COUNT(*) AS liczba
FROM lokalizacje
WHERE planeta_id = @kuat_planeta_id
GROUP BY nazwa
HAVING COUNT(*) > 1;


-- 6. KONTROLA BRAKU RODZICA DLA GŁÓWNYCH KORZENI

SELECT
    nazwa,
    typ,
    rodzic_id
FROM lokalizacje
WHERE planeta_id = @kuat_planeta_id
  AND nazwa IN (
      'Kuat — Powierzchnia Planety',
      'Kuat — Orbita'
  );


-- 7. PEŁNA HIERARCHIA KUAT

WITH RECURSIVE hierarchia AS
(
    SELECT
        id,
        rodzic_id,
        nazwa,
        typ,
        0 AS poziom,
        CAST(nazwa AS CHAR(4000)) AS sciezka
    FROM lokalizacje
    WHERE planeta_id = @kuat_planeta_id
      AND rodzic_id IS NULL

    UNION ALL

    SELECT
        dziecko.id,
        dziecko.rodzic_id,
        dziecko.nazwa,
        dziecko.typ,
        rodzic.poziom + 1,
        CONCAT(rodzic.sciezka, ' > ', dziecko.nazwa)
    FROM lokalizacje dziecko
    INNER JOIN hierarchia rodzic
        ON dziecko.rodzic_id = rodzic.id
    WHERE dziecko.planeta_id = @kuat_planeta_id
)

SELECT
    poziom,
    typ,
    nazwa,
    sciezka
FROM hierarchia
ORDER BY sciezka;


-- 8. KOMPLETNOŚĆ POWIERZCHNIOWYCH BIUR KDY

SELECT
    nazwa,
    typ,
    rodzaj,
    publiczna,
    status,
    rodzic_id
FROM lokalizacje
WHERE planeta_id = @kuat_planeta_id
  AND nazwa LIKE 'Kuat Drive Yards —%'
ORDER BY nazwa;


-- 9. KONTROLA OBECNOŚCI OBIEKTÓW KDY

SELECT
    nazwa,
    CASE
        WHEN COUNT(*) > 0 THEN 'OBECNY'
        ELSE 'BRAK'
    END AS wynik
FROM
(
    SELECT 'Kompleks Administracyjny Kuat Drive Yards' AS nazwa
    UNION ALL SELECT 'Kuat Drive Yards — Biura Zarządu'
    UNION ALL SELECT 'Kuat Drive Yards — Główna Administracja Korporacyjna'
    UNION ALL SELECT 'Kuat Drive Yards — Biuro Projektowe'
    UNION ALL SELECT 'Kuat Drive Yards — Centrum Inżynieryjne'
    UNION ALL SELECT 'Kuat Drive Yards — Dział Dokumentacji Technicznej'
    UNION ALL SELECT 'Kuat Drive Yards — Archiwum Projektowe'
    UNION ALL SELECT 'Kuat Drive Yards — Centrum Badawczo-Rozwojowe'
    UNION ALL SELECT 'Kuat Drive Yards — Centrum Logistyczne'
    UNION ALL SELECT 'Kuat Drive Yards — Placówka Ochrony Korporacyjnej'
    UNION ALL SELECT 'Kuat — Andrim Shipyards'
    UNION ALL SELECT 'Kuat — Deponn Shipyards'
    UNION ALL SELECT 'Kuat — Maw Shipyards'
    UNION ALL SELECT 'Kuat Drive Yards — Zakłady Produkcji Komponentów'
) wymagane
LEFT JOIN lokalizacje l
    ON l.planeta_id = @kuat_planeta_id
   AND l.nazwa = wymagane.nazwa
GROUP BY wymagane.nazwa
ORDER BY wymagane.nazwa;


-- 10. KONTROLA ROZDZIELENIA POWIERZCHNI I ORBITY

SELECT
    CASE
        WHEN rodzic.nazwa = 'Kuat — Orbita'
          OR dziecko.nazwa = 'Kuat — Orbita'
          OR rodzic.nazwa = 'Kuat Drive Yards Orbital Array'
          OR dziecko.nazwa = 'Kuat Drive Yards Orbital Array'
          OR dziecko.nazwa LIKE 'Kuat — Andrim%'
          OR dziecko.nazwa LIKE 'Kuat — Deponn%'
          OR dziecko.nazwa LIKE 'Kuat — Maw%'
        THEN 'INFRASTRUKTURA ORBITALNA'
        ELSE 'INFRASTRUKTURA POWIERZCHNIOWA'
    END AS srodowisko,
    COUNT(*) AS liczba
FROM lokalizacje dziecko
LEFT JOIN lokalizacje rodzic
    ON rodzic.id = dziecko.rodzic_id
WHERE dziecko.planeta_id = @kuat_planeta_id
GROUP BY srodowisko;


-- 11. KONTROLA WSPÓŁRZĘDNYCH

SELECT
    COUNT(*) AS liczba_lokalizacji,
    SUM(x_km IS NULL) AS brak_x,
    SUM(y_km IS NULL) AS brak_y
FROM lokalizacje
WHERE planeta_id = @kuat_planeta_id;


-- 12. KOŃCOWY STATUS IMPORTU

SELECT
    CASE
        WHEN EXISTS (
            SELECT 1
            FROM lokalizacje
            WHERE planeta_id = @kuat_planeta_id
              AND nazwa = 'Kuat — Powierzchnia Planety'
        )
        AND EXISTS (
            SELECT 1
            FROM lokalizacje
            WHERE planeta_id = @kuat_planeta_id
              AND nazwa = 'Kuat — Orbita'
        )
        AND EXISTS (
            SELECT 1
            FROM lokalizacje
            WHERE planeta_id = @kuat_planeta_id
              AND nazwa = 'Kuat Drive Yards — Biura Zarządu'
        )
        AND EXISTS (
            SELECT 1
            FROM lokalizacje
            WHERE planeta_id = @kuat_planeta_id
              AND nazwa = 'Kuat — Andrim Shipyards'
        )
        AND EXISTS (
            SELECT 1
            FROM lokalizacje
            WHERE planeta_id = @kuat_planeta_id
              AND nazwa = 'Kuat — Deponn Shipyards'
        )
        AND EXISTS (
            SELECT 1
            FROM lokalizacje
            WHERE planeta_id = @kuat_planeta_id
              AND nazwa = 'Kuat — Maw Shipyards'
        )
        AND NOT EXISTS (
            SELECT 1
            FROM lokalizacje dziecko
            LEFT JOIN lokalizacje rodzic
                ON rodzic.id = dziecko.rodzic_id
            WHERE dziecko.planeta_id = @kuat_planeta_id
              AND dziecko.rodzic_id IS NOT NULL
              AND rodzic.id IS NULL
        )
        THEN 'KUAT: IMPORT I HIERARCHIA POPRAWNE'
        ELSE 'KUAT: WYKRYTO BRAKI — WYMAGANA KONTROLA'
    END AS status_koncowy;
